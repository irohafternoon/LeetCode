# 63. Unique Paths II

## STEP1
- 何も見ずに解いてみる

#### 考えたこと
- 前問のUnique Path(障害物がない)とほぼ同様にDPを利用することでできそう
- そのマスが障害物である場合は、その場所のDPは更新しない（初期値0のまま）
- スタート地点が障害物の場合は、常に0通りであることに注意

計算量
- 時間計算量 O(M * N) M = N = 100の時　 10マイクロ秒と見積もり
- 空間計算量 O(M * N)

```cpp
class Solution {
public:
    int uniquePathsWithObstacles(std::vector<std::vector<int>>& obstacleGrid) {
        if (obstacleGrid.empty() || obstacleGrid[0].empty()) {
            return 0;
        }
        if (obstacleGrid[0][0] == kObstacle) {
            return 0;
        }
        int num_rows = obstacleGrid.size();
        int num_columns = obstacleGrid[0].size();
        std::vector<std::vector<int>> coordinate_to_paths(num_rows,
                                                          std::vector<int>(num_columns));
        coordinate_to_paths[0][0] = 1;
        for (int row = 0; row < num_rows; row++) {
            for (int column = 0; column < num_columns; column++) {
                if (obstacleGrid[row][column] == kObstacle) {
                    continue;
                }
                if (row > 0) {
                    coordinate_to_paths[row][column] += coordinate_to_paths[row - 1][column];
                }
                if (column > 0) {
                    coordinate_to_paths[row][column] += coordinate_to_paths[row][column - 1];
                }
            }
        }
        return coordinate_to_paths.back().back();
    }
private:
    static constexpr int kObstacle = 1;
};
```

class Solution {
public:
    int uniquePathsWithObstacles(std::vector<std::vector<int>>& obstacleGrid) {
        if (obstacleGrid.empty() || obstacleGrid[0].empty()) {
            return 0;
        }
        if (obstacleGrid[0][0] == kObstacle) {
            return 0;
        }
        int num_rows = obstacleGrid.size();
        int num_columns = obstacleGrid[0].size();
        std::vector<std::vector<int>> coordinte_to_paths(num_rows,
                                                         std::vector<int>(num_columns));
        coordinte_to_paths[0][0] = 1;
        for (int row = 0; row < num_rows; row++) {
            for (int column = 0; column < num_columns; column++) {
                if (obstacleGrid[row][column] == kObstacle) {
                    continue;
                }
                if (row > 0) {
                    coordinte_to_paths[row][column] += coordinte_to_paths[row - 1][column];
                }
                if (column > 0) {
                    coordinte_to_paths[row][column] += coordinte_to_paths[row][column - 1];
                }
            }
            if (*std::max_element(coordinte_to_paths[row].begin(),coordinte_to_paths[row].end()) == 0) {
                return 0;
            }
        }
        return coordinte_to_paths.back().back();
    }
private:
    static constexpr int kObstacle = 1;
};
