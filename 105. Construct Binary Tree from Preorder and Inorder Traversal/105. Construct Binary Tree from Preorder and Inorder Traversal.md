## STEP1

#### 考えたこと
- preorderはノードの左右の順序について満たすべき制約を規定しているので、再帰関数でpreorder順にみて、ノードの場所が制約を満たしている箇所では、preorderの値を順に置いていく


計算量
- 時間計算量 O(N) ステップ数は辺と頂点で 3Nステップ N = 3000の時 9.0 * 10^3ステップ 9マイクロ秒と見積もり

- 空間計算量 ワーストケースでO(N)

```cpp
class Solution {
public:
    TreeNode* buildTree(vector<int>& preorder, vector<int>& inorder) {
        std::map<int, int> value_to_inorder_index;
        for (int i = 0; i < inorder.size(); i++) {
            value_to_inorder_index[inorder[i]] = i;
        }
        int preorder_index = 0;
        return BuildTreeHelper(preorder, 0, inorder.size() - 1,
                               value_to_inorder_index, preorder_index);
    }
private:
    TreeNode* BuildTreeHelper(vector<int>& preorder, int left, int right, 
                              const std::map<int, int>& value_to_inorder_index,
                              int& preorder_index) {
        if (preorder_index >= preorder.size()) { 
            return nullptr;
        }
        int value = preorder[preorder_index];
        int inorder_index = value_to_inorder_index.at(value);
        if (!(left <= inorder_index && inorder_index <= right)) {
            return nullptr;
        }
        TreeNode* node = new TreeNode(value);
        ++preorder_index;
        node->left = BuildTreeHelper(preorder,left, inorder_index - 1,
                                     value_to_inorder_index, preorder_index);
        node->right = BuildTreeHelper(preorder, inorder_index + 1, right,
                                      value_to_inorder_index, preorder_index);
        return node;
    }
};
```

非再帰でのDFS(pre-order順)
```cpp
class Solution {
public:
    TreeNode* buildTree(vector<int>& preorder, vector<int>& inorder) {
        std::map<int, int> value_to_inorder_index;
        for (int i = 0; i < inorder.size(); i++) {
            value_to_inorder_index[inorder[i]] = i;
        }
        std::stack<std::tuple<TreeNode**, int, int>> nodes_to_visit;
        TreeNode* root = nullptr;
        nodes_to_visit.emplace(&root, 0, inorder.size() - 1);
        int preorder_index = 0;
        while (!nodes_to_visit.empty()) {
            auto [ptr_to_node, inorder_left_limit, inorder_right_limit] = nodes_to_visit.top();
            nodes_to_visit.pop();
            int value = preorder[preorder_index];
            int node_position = value_to_inorder_index[value];
            if (!(inorder_left_limit <= node_position && node_position <= inorder_right_limit)) {
                continue;
            }
            *ptr_to_node = new TreeNode(value);
            TreeNode* node = *ptr_to_node;
            ++preorder_index;
            if (preorder_index == preorder.size()) {
                break;
            }
            // DFS順に探索するため、stackにright→leftの順番でstackに積む
            nodes_to_visit.emplace(&(node->right), node_position + 1, inorder_right_limit);
            nodes_to_visit.emplace(&(node->left), inorder_left_limit, node_position - 1);
        }
        return root;
    }
};
```

```cpp
class Solution {
public:
    TreeNode* buildTree(vector<int>& preorder, vector<int>& inorder) {
        std::map<int, int> value_to_inorder_index;
        for (int i = 0; i < inorder.size(); i++) {
            value_to_inorder_index[inorder[i]] = i;
        }
        int preorder_index = 0;
        return BuildTreeHelper(preorder, 
                               0, preorder.size() - 1, 
                               0, inorder.size() - 1,
                               value_to_inorder_index);
    }
private:
    TreeNode* BuildTreeHelper(vector<int>& preorder,
                              int pre_left, int pre_right,
                              int in_left, int in_right,
                              const std::map<int, int>& value_to_inorder_index) {
        if (!(pre_left <= pre_right) || !(in_left <= in_right)) { 
            return nullptr;
        }
        int value = preorder[pre_left];
        TreeNode* node = new TreeNode(value);
        int in_mid = value_to_inorder_index.at(value);
        int num_left_nodes = in_mid - in_left;
        int num_right_nodes = in_right - in_mid;
        //左部分木は、pre_indexではpre_left+1から始めてnum_left_nodes個分ある
        node->left = BuildTreeHelper(preorder, 
                                     pre_left + 1, pre_left + num_left_nodes,
                                     in_left, in_mid - 1,
                                     value_to_inorder_index);
        //右部分木は、pre_indexではpre_rightから始めてnum_right_nodes個分戻ったところまで
        node->right = BuildTreeHelper(preorder,
                                      pre_right - num_right_nodes + 1, pre_right,
                                      in_mid + 1, in_right,
                                      value_to_inorder_index);
        return node;
    }
};

```

- inorder順に走査して、preorderの制限を確認しながらnodeを埋めていく方法　https://github.com/tarinaihitori/leetcode/pull/29/files#r2044913447
- とりあえず理解したポイント
    - inorder順にノードを見るとき、必ず、今見たnodeのleftはあるなら必ず既に見ている
    - in-order順に見て、まだ右の子が確定していないノードはstackに入れる
    - あるノードを見た時、それがstackに入っているノードよりpre-orderで前にある場合は、そのノードよりpre-order順で後のものは回収されるはず
    - stackのノードは全て.leftが埋まっているのだから、stackをpopする順に.rightで連鎖してつなげていく
    - 今見たノードの右側のノードというのはまだ登場しているはずがないのだから(=stackのノードは、今見たノードの必ず左部分木に属するから)、今見たノードの.leftに連鎖してつなげたものを連結する
    - この操作を続けると、最後に完成した木の一番右端のラインにあるノード(root含む)がstackに残るので、全て回収する。

- 自分でこれを初見で書くというのは厳しいが、とりあえず上記くらいは紙に書いたりして把握した。

```cpp
class Solution {
public:
    TreeNode* buildTree(std::vector<int>& preorder, std::vector<int>& inorder) {
        std::map<int, int> value_to_preorder_index;
        for (int i = 0; i < preorder.size(); i++) {
            value_to_preorder_index[preorder[i]] = i;
        }
        std::stack<TreeNode*> nodes_right_unfixed;
        for (auto value : inorder) {
            TreeNode* node = new TreeNode(value);
            int node_position = value_to_preorder_index[value];
            TreeNode* child = nullptr;
            while (!nodes_right_unfixed.empty()) {
                TreeNode* parent_candidate = nodes_right_unfixed.top();
                if (value_to_preorder_index[parent_candidate->val] < node_position) {
                    break;
                }
                nodes_right_unfixed.pop();
                parent_candidate->right = child;
                child = parent_candidate;
            }
            node->left = child;
            nodes_right_unfixed.push(node);
        }
        //一番右端のラインにあるノード（root含む）を回収するので、最終的なchildがrootになる
        TreeNode* child = nullptr;
        while (!nodes_right_unfixed.empty()) {
            TreeNode* parent = nodes_right_unfixed.top();
            nodes_right_unfixed.pop();
            parent->right = child;
            child = parent; 
        }
        return child;
    }
};
```

```cpp
class Solution {
public:
    TreeNode* buildTree(std::vector<int>& preorder, std::vector<int>& inorder) {
        std::map<int, int> value_to_preorder_index;
        for (int i = 0; i < preorder.size(); i++) {
            value_to_preorder_index[preorder[i]] = i;
        }
        std::stack<TreeNode*> nodes_right_unfixed;
        for (auto value : inorder) {
            int node_position = value_to_preorder_index[value];
            TreeNode* node = new TreeNode(value);
            node->left = gather_descendants(node_position, value_to_preorder_index,
                                            nodes_right_unfixed);
            nodes_right_unfixed.push(node);
        }
        return gather_descendants(-1, value_to_preorder_index, nodes_right_unfixed);
    }
private:
    TreeNode* gather_descendants(int node_position,
                                 const std::map<int, int>& value_to_preorder_index,
                                 std::stack<TreeNode*>& nodes_right_unfixed) {
        TreeNode* child = nullptr;
        while (!nodes_right_unfixed.empty()) {
            TreeNode* parent_candidate = nodes_right_unfixed.top();
            if (value_to_preorder_index.at(parent_candidate->val) < node_position) {
                break;
            }
            nodes_right_unfixed.pop();
            parent_candidate->right = child;
            child = parent_candidate;
        }
        return child;
    }
};
```
